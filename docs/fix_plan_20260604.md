def load_config(path: str = "config/settings.yaml") -> dict:
    """单例配置加载, 返回嵌套 dict。
    env var 展开: ${VAR_NAME} -> os.environ['VAR_NAME']
    """
    import os, yaml
    p = Path(path)
    if not p.exists():
        return {}
    raw = p.read_text(encoding="utf-8")
    # env var 展开
    raw = re.sub(r"\$\{(\w+)\}", lambda m: os.environ.get(m.group(1), m.group(0)), raw)
    return yaml.safe_load(raw) or {}

def get(cfg: dict, *keys, default=None):
    """安全 nested get: get(cfg, 'risk', 'max_daily_loss_pct', default=5.0)"""
    for k in keys:
        if not isinstance(cfg, dict) or k not in cfg:
            return default
        cfg = cfg[k]
    return cfg if cfg is not None else default
```

### 修法 diff

# risk/position.py:62-L65  -- 当前 buggy 代码

def update_on_bar(self, bid: float, ask: float, atr: float = None) -> None:
    if atr is None:
        atr = 1.0  # 防御默认值, 上游应传
    if self._direction == 1:  # long
        self._trail_high = max(self._trail_high, bid)
        if self._atr_mult is not None:
            new_sl = self._trail_high - self._atr_mult * atr
            if new_sl > self._sl_price:  # SL 只能上调
                self._sl_price = new_sl
    elif self._direction == -1:  # short
        self._trail_low = min(self._trail_low, ask)
        if self._atr_mult is not None:
            new_sl = self._trail_low + self._atr_mult * atr
            if new_sl < self._sl_price:
                self._sl_price = new_sl
```

def on_fill(self, order):
    if state.position.direction == 0:
        # 首笔子单, 初始化
        state.position.direction = order.direction
        state.position.volume = order.volume
        state.position.entry_price = order.fill_price

            (state.position.entry_price * state.position.volume +

        state.position.volume = new_vol
    else:
        # 反向减仓或翻仓, 单独处理 (本期不改, 留 TODO)

**预计总耗时**: 1 个完整工作日(8h),每个 fix 10-15 分钟。

## 📊 风险评估矩阵

| Phase | 风险 | 缓解 |
|---|---|---|
| Phase 1 (P1-P8) | 改架构面广,可能 break 现有功能 | 每个 P 独立 commit,独立 test;P1 后跑 backtest 冒烟 |
| Phase 1 P5 (state lock) | 改 60+ 调用点,容易漏 | 拆 5a/5b 两步走,观察期 |
| Phase 2 batch | 同模块多个 fix 一起,debug 难定位 | 选 small/medium 风险同 batch,大改动单独 |
| 全量回归 | 可能发现旧 test 跟新口径不一致 | 旧 test 改用 cfg 注入,不再 hardcode 数字 |

## ✅ 完成定义 (DoD)

每个 fix commit 必须满足:
- [ ] 改动的代码通过 `pytest tests/test_<id>.py -v`
- [ ] 整 repo 跑 `pytest tests/ -v` 不退化
- [ ] backtest 跑 1 次,PnL/Sharpe 变化 < 5%(纯 fix 不应有大变化)
- [ ] commit message 引用 audit 报告 ID,例如 `fix(BUG-6): trailing SL formula uses ATR not (sl-entry) sign`
- [ ] `git log --oneline` 50 commit 内可全部 `git revert`
